### Test 4: model output equivalence (NOT YET DONE)
Prove that pol_1858 from the new model == pol_4288[:, sl_idx] from the old model.

The correct script is SIMPLE -- previous session wasted time on subprocess complexity
and accidentally ran CPU inference (slow). The right approach:

```python
import sys, pickle, gc
import numpy as np
import torch

sys.path.insert(0, r"C:\Users\Bryan\repos\chessbot\src")
import pyfastchess
from chessbot.model import PT_BUILDERS, VARIANTS

SNAPSHOT = r"C:\Users\Bryan\Data\chessbot_data\selfplay_runs\precond_run2\grad_analysis_snapshot.pkl"
PT_PATH  = r"C:\Users\Bryan\Data\chessbot_data\selfplay_runs\16m_precond_run0\16m_precond_run0_model.pt"
DEVICE   = torch.device("cuda")

sl_idx = np.where(pyfastchess.build_sometimes_legal_mask())[0]  # (1858,)

with open(SNAPSHOT, "rb") as f:
    data = pickle.load(f)
tokens_np = np.stack([data[i][0] for i in range(1024)]).astype(np.int32)
tokens_t  = torch.from_numpy(tokens_np).long().to(DEVICE)

# model A: 4288
obj = torch.load(PT_PATH, map_location=DEVICE)
arch = obj.get("arch", "16m-precond-smartgate")
modelA = PT_BUILDERS[arch](VARIANTS[arch], policy_1858=False).to(DEVICE).eval()
modelA.load_state_dict(obj["model"], strict=False)
with torch.no_grad():
    polA, _ = modelA(tokens_t)
pol4288 = polA.float().cpu().numpy()
del modelA, obj; gc.collect(); torch.cuda.empty_cache()

# model B: 1858
obj = torch.load(PT_PATH, map_location=DEVICE)
modelB = PT_BUILDERS[arch](VARIANTS[arch], policy_1858=True).to(DEVICE).eval()
modelB.load_state_dict(obj["model"], strict=False)
with torch.no_grad():
    polB, _ = modelB(tokens_t)
pol1858 = polB.float().cpu().numpy()
del modelB, obj; gc.collect(); torch.cuda.empty_cache()

diff = np.abs(pol1858 - pol4288[:, sl_idx])
print(f"max diff: {diff.max():.2e}, mean diff: {diff.mean():.2e}")
assert diff.max() < 1e-4, "FAIL"
print("PASS")
```

Notes on the snapshot data structure:
    data = list of tuples: (tokens, pol_target, pol_4288, wdl, outcome, weight)
    tokens: int32 (64,) -- board encoding, direct model input
    pol_4288: float32 (4288,) -- stored NN output logits (nonzero at ~4 positions)

sl_idx comes from pyfastchess.build_sometimes_legal_mask() -- returns bool array (4288,),
np.where gives the (1858,) indices of the sometimes-legal slots.
